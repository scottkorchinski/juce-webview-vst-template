cmake_minimum_required(VERSION 3.22)

set(TEMPLATE_CONFIG_PATH "${CMAKE_CURRENT_SOURCE_DIR}/template.config.json")
if(NOT EXISTS "${TEMPLATE_CONFIG_PATH}")
  message(FATAL_ERROR "Missing template config: ${TEMPLATE_CONFIG_PATH}")
endif()

file(READ "${TEMPLATE_CONFIG_PATH}" TEMPLATE_CONFIG_JSON)

string(JSON TEMPLATE_PROJECT_NAME GET "${TEMPLATE_CONFIG_JSON}" projectName)
string(JSON TEMPLATE_PRODUCT_NAME GET "${TEMPLATE_CONFIG_JSON}" productName)
string(JSON TEMPLATE_COMPANY_NAME GET "${TEMPLATE_CONFIG_JSON}" companyName)
string(JSON TEMPLATE_BUNDLE_ID GET "${TEMPLATE_CONFIG_JSON}" bundleId)
string(JSON TEMPLATE_PLUGIN_CODE GET "${TEMPLATE_CONFIG_JSON}" pluginCode)
string(JSON TEMPLATE_MANUFACTURER_CODE GET "${TEMPLATE_CONFIG_JSON}" manufacturerCode)
string(JSON TEMPLATE_IS_SYNTH GET "${TEMPLATE_CONFIG_JSON}" isSynth)
string(JSON TEMPLATE_EDITOR_WIDTH GET "${TEMPLATE_CONFIG_JSON}" editorWidth)
string(JSON TEMPLATE_EDITOR_HEIGHT GET "${TEMPLATE_CONFIG_JSON}" editorHeight)
string(JSON TEMPLATE_DEV_SERVER_URL GET "${TEMPLATE_CONFIG_JSON}" devServerUrl)

string(JSON TEMPLATE_FORMATS_LENGTH LENGTH "${TEMPLATE_CONFIG_JSON}" formats)
math(EXPR TEMPLATE_FORMATS_LAST_INDEX "${TEMPLATE_FORMATS_LENGTH} - 1")
set(TEMPLATE_FORMATS "")
foreach(FORMAT_INDEX RANGE ${TEMPLATE_FORMATS_LAST_INDEX})
  string(JSON TEMPLATE_FORMAT_VALUE GET "${TEMPLATE_CONFIG_JSON}" formats ${FORMAT_INDEX})
  list(APPEND TEMPLATE_FORMATS "${TEMPLATE_FORMAT_VALUE}")
endforeach()

project(${TEMPLATE_PROJECT_NAME} VERSION 0.1.0)

set(CMAKE_CXX_STANDARD 20)
if(APPLE)
  # Build universal binaries so Apple Silicon (arm64) and Intel (x86_64) hosts
  # can both see and load the plugin.
  set(CMAKE_OSX_ARCHITECTURES "x86_64;arm64" CACHE STRING "" FORCE)
endif()
set(LIB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/libs)
include(cmake/cpm.cmake)

CPMAddPackage(
  NAME JUCE
  GIT_TAG 8.0.6
  VERSION 8.0.6
  GITHUB_REPOSITORY juce-framework/JUCE
  SOURCE_DIR ${LIB_DIR}/juce
)

if(MSVC)
  set(CXX_PROJECT_WARNINGS "/W4;/WX;/wd4820;/wd4514")
else()
  set(CXX_PROJECT_WARNINGS "-Wall;-Werror;-Wextra;-Wpedantic")
endif()

add_subdirectory(plugin)
